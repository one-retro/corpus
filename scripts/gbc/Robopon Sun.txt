# Robopon - Sun Version (HuC-3: clock + 32 KiB battery RAM): New Game, mash A
# through the long opening (the name becomes "AAAA", birthday Jan. 01) until
# Grandpa Hogle's lab, where talking to him loops; walk right, away from him,
# then start menu -> Save -> "Overwrite existing file?" Yes -> "Saved!".
wait 1200
repeat 300          # title, New Game, the opening, name and birthday, the lab;
tap a 1 40          # surplus presses only replay Grandpa's lines
end
repeat 16           # A ends his lines; RIGHT is ignored while he talks and
tap a 1 40          # walks away from him once he stops, so the next A has
press right 10      # no one to talk to
wait 10
end
wait 30
press start 6       # Robopon / Tools / Garage / Data / Settings / Save / Title
wait 120            # (the menu fades in)
repeat 5
press down 6
wait 20
end
press a 6           # Save -> "Overwrite existing file?" Yes / No
wait 60
press a 6           # Yes -> Please Wait -> Saved!
wait 120
shot saved          # "Saved!"
wait 30
