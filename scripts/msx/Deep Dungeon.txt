# Deep Dungeon: register the name "ABC" (なまえとうろく). The game writes the
# roster (name, HP, level) to its 8 KiB SRAM. Joystick only: fMSX runs with
# the port set to "Joystick + Emulated Keyboard" (--device0 257).
# The reset is the power cycle (blueMSX writes its .sram only then); after it
# the roster shows ABC again.
wait 600
press a 8           # title -> roster
wait 150
repeat 5
press down 6        # -> なまえとうろく (register)
wait 20
end
press a 8           # -> pick a slot (slot 1)
wait 100
press a 8           # -> name grid, cursor on あ
wait 150
repeat 3
press down 6
wait 20
end
repeat 6
press right 6       # -> A
wait 20
end
press a 6
wait 30
press right 6
wait 20
press a 6           # B
wait 30
press right 6
wait 20
press a 6           # C
wait 30
press b 8           # -> おわり (done)
wait 60
press a 8
wait 200
shot registered
reset
wait 600
press a 8
wait 200
shot roster
