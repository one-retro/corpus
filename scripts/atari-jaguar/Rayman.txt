# Rayman: create save slot 1. On an empty slot B makes the file at once
# (default name AAA, 3 lives, 5 continues, 0%), and the game writes it to
# the cartridge EEPROM. Then reset and come back to PLAYER SELECT, which
# reads the slot back from the EEPROM.
wait 900
press start 6       # skip the Ubi Soft logo
wait 300
press b 6           # title -> intro text
wait 300
press start 6       # -> PLAYER SELECT
wait 300
shot empty
press b 6           # slot 1: the file is created and saved here
wait 240
shot created
reset
wait 900
press start 6
wait 300
press b 6
wait 300
press start 6
wait 300
shot reloaded
