# Famicom Grand Prix II - 3D Hot Rally (Disk Writer): GAME SELECT > NAME
# opens the Disk Writer service form; type アイウ and choose "save", which
# writes TDREDATA (side A). The console is then reset and the form, read back
# from the disk, shows the name.
wait 1200
press start 10      # title -> GAME SELECT
wait 400
press down 10
wait 90
press down 10       # NAME
wait 90
press start 10
wait 600
include _service-form.inc
tap right 12 12     # save
shot name
press a 6           # save: TDREDATA is written
wait 1500
shot written
reset
wait 1200
press start 10
wait 400
press down 10
wait 90
press down 10
wait 90
press start 10      # the form as the disk holds it
wait 600
shot saved
wait 30
