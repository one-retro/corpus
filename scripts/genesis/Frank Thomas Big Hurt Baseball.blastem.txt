# first boot, which is not a save. Reset and open Season: Season Type
# is now Resume, and resuming opens Baltimore's game day. Genesis A/B/C are
# the RetroPad's y/b/a (any of them confirms here).

press r 6

press right 6       # Yes
wait 60
press r 6
wait 120            # Are You Sure?: No
press right 6       # Yes
wait 60
press r 6
wait 500            # Season Adjusted 95, Length 26 (Short), Play Mode Regulation
press r 6
wait 500            # TEAM SELECT
press r 6           # Baltimore
wait 500            # GAME DAY: Baltimore vs New York AMR (season saved)

press r 6
