load corpus-tmp/mgl/turbogolf.mgl 8
# Jack Nicklaus' Turbo Golf: skins play with the default players, tee off on
# hole 1, then Select+Run: "Head back to the clubhouse?" YES, "Save this
# Game?" YES writes JGF-SG.001 to backup RAM. After a reset the title asks
# "Resume Saved Game?".
repeat 8
wait 120
press start 6         # title, play mode, players, main menu
end
tap a 8 90            # defaults through to the first tee (the last presses swing: out of bounds)
wait 300
press start 6         # "Out of bounds! 1 stroke penalty. Press any Button."
wait 60
press select 6
wait 60
press b 6
wait 60
shot tee
press select+start 6  # Head back to the clubhouse?
wait 90
press a 6             # YES
wait 90
shot savegame         # Save this Game?
press a 6             # YES
wait 300
wait 30
save
wait 60
shot final
# power cycle: load the game again; the core reads the .sav back from the SD card
load corpus-tmp/mgl/turbogolf.mgl 8
wait 120
press start 6
wait 180
shot resume           # Resume Saved Game?  Yes or No
