load corpus-tmp/mgl/stadiumtpak.mgl 12
# MiSTer variant: a shot of the Game Pak Check (Transfer Pak with Pokemon Red).
# Pokemon Stadium: OPTIONS -> SOUND set to MONO. The game formats its
# FlashRAM on the first boot and writes the setting when the options close.
# The script then resets and opens OPTIONS again to show MONO was read back.
# No Transfer Pak is plugged in ("Game Pak None" on all four ports).
# N64 buttons on the RetroPad: A = b, B = y, Z = l2, Start = start.
wait 2000
press start 6    # title -> Game Pak Check
wait 200
shot pakcheck
press a 6    # OK -> Please Select
wait 250
press down 6
wait 30
press right 6    # OPTIONS
wait 30
press a 6
wait 250
press a 6    # SOUND: STEREO -> MONO
wait 150
shot mono
tap down 3 20    # QUIT
wait 20
press a 6    # close the options: the setting is written here
wait 250
wait 30
save
wait 60
shot final
# power cycle: load the game again; the core reads the .sav back from the SD card
load corpus-tmp/mgl/stadiumtpak.mgl 12
wait 2000
press start 6
wait 200
press a 6
wait 250
press down 6
wait 30
press right 6
wait 30
press a 6    # OPTIONS again after the reset
wait 250
shot reloaded
