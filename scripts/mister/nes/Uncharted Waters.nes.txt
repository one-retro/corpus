# New Adventure: hero "AAA", the rolled stats accepted, every bonus point spent (the roll is random
# and differs between cores, so each stat gets more presses than it can take: once the bonus is 0 the
# game asks "Is this OK?", the next A says yes and the rest type into the flagship's name). The story,
# then Lisbon: SELECT -> Save -> Y, Quit game? -> N (RIGHT).

wait 600

press start
wait 300
press a               # name OK? Y
wait 600
press a               # stats OK? Y (RIGHT would reroll)
wait 600
shot bonus

tap a 70 16
press down
wait 30
end
press start           # flagship name done
wait 400
press a               # name OK? Y
wait 600

press a

end

wait 600
