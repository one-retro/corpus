# Family Circuit '91 (Namco 175, 2 KiB of battery RAM): SETTING > GARAGE >
# SAVE, slot A, name the setting "ABBBBB": the car setting is stored in the
# battery RAM. GARAGE > LOAD then lists slot A as ABBBBB.
# The title ignores input for its first ~1000 frames and takes A, not Start.
wait 1500
press a 10          # title -> SELECT MODE
wait 300
press a 10          # SETTING
wait 300
press right 10      # GARAGE
wait 60
press a 10
wait 300
press a 10          # SAVE
wait 300
press a 10          # slot A -> INPUT NAME
wait 300
press a 10          # A
wait 30
press right 10
wait 30
tap a 5 30          # BBBBB: the sixth letter ends the name
wait 200
shot name
press a 10          # the setting is shown, then stored
wait 200
press a 10          # back to GARAGE
wait 200
press down 10       # LOAD
wait 60
press a 10
wait 300
shot load           # slot A: ABBBBB
press b 10
wait 60
