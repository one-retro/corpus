press start 6       # title -> SELECT (cursor already on REGISTER YOUR NAME)
wait 120
press start 6       # REGISTER YOUR NAME
wait 90

repeat 11
tap right 1 8
end
tap a 1 20          # L
tap left 3 8
tap a 1 20          # I
tap right 5 8
tap a 1 20          # N
tap left 3 8
tap a 1 20          # K
shot name
press select 6      # heart: slot 2
wait 20
press select 6      # slot 3
wait 20
press select 6      # REGISTER END
wait 20
press start 6       # register: the file is written
wait 120
