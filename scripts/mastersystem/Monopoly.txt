# Monopoly: two human players "A" and "B", no computer player, no time limit,
# roll for first turn, then MAIN MENU > MORE OPTIONS > SAVE GAME:
# "GAME SAVED. PRESS B1 TO CONTINUE". Button 2 (a) starts from the title and
# begins the game; button 1 (b) enters everything else. Menu moves are spaced
# 30 frames apart: closer presses skip a line.
wait 400
press a 10          # PRESS START BUTTON -> CHOOSE A TOKEN
wait 150
press b 6           # first token -> ENTER YOUR NAME
wait 120
press b 6           # A
wait 60
press a 6           # -> DONE
wait 60
press b 6           # DONE: player 1 "A"
wait 150
press right 6       # next token
wait 60
press b 6           # -> ENTER YOUR NAME
wait 150
press right 6       # -> B
wait 60
press b 6           # B
wait 60
press a 6           # -> DONE
wait 60
press b 6           # DONE: player 2 "B"
wait 250
press a 6           # B-2 begin: COMPUTER PLAYER? NO
wait 150
press b 6           # NO -> SET TIME LIMIT? NO
wait 150
press b 6           # NO -> ROLL DICE TO DECIDE WHO STARTS FIRST
wait 250
press b 6           # roll; the other player rolls, SUMMARY OF PLAYERS
wait 800
press b 6           # -> MAIN MENU, MENU OPEN TO (first player)
wait 250
repeat 6            # ROLL DICE ... DISPLAY DEEDS -> MORE OPTIONS
press down 6
wait 30
end
press b 6
wait 150
repeat 4            # SET SCROLL / TIMER / MUSIC / CONVERT PLAYER -> SAVE GAME
press down 6
wait 30
end
press b 6           # "GAME SAVED"
wait 120
shot saved
wait 30
